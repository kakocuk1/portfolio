Пример моего кода

Kafka Consumer с retry-логикой и graceful shutdown из Report Service

Go • Kafka • Concurrency 
package consumer

import (
    "context"
    "log"
    "time"
    "github.com/segmentio/kafka-go"
)

// ReportConsumer обрабатывает сообщения из Kafka
type ReportConsumer struct {
    reader     *kafka.Reader
    handler    ReportHandler
    maxRetries int
}

// Consume запускает цикл обработки с graceful shutdown
func (c *ReportConsumer) Consume(ctx context.Context) error {
    defer c.reader.Close()

    for {
        select {
        case <-ctx.Done():
            log.Println("Graceful shutdown: stopping consumer")
            return ctx.Err()
            
        default:
            msg, err := c.reader.FetchMessage(ctx)
            if err != nil {
                log.Printf("Error fetching message: %v", err)
                continue
            }

            // Обрабатываем с retry-логикой
            if err := c.processWithRetry(ctx, msg); err != nil {
                log.Printf("Failed after %d retries: %v", c.maxRetries, err)
                continue
            }

            // Коммитим offset только после успеха
            c.reader.CommitMessages(ctx, msg)
        }
    }
}

// processWithRetry обрабатывает сообщение с экспоненциальной задержкой
func (c *ReportConsumer) processWithRetry(ctx context.Context, msg kafka.Message) error {
    var lastErr error
    
    for attempt := 1; attempt <= c.maxRetries; attempt++ {
        err := c.handler.Handle(ctx, msg.Value)
        if err == nil {
            return nil
        }
        
        lastErr = err
        backoff := time.Duration(attempt*attempt) * time.Second
        
        select {
        case <-ctx.Done():
            return ctx.Err()
        case <-time.After(backoff):
        }
    }
    
    return lastErr
}
